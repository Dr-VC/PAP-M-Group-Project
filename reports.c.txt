#include <stdio.h>
#include "employees.h"
#include "budget.h"
#include "suppliers.h"
#include "assets.h"
#include "reports.h"

void generateEmployeeReport(void) {
    printf("\n========================================\n");
    printf("          EMPLOYEE SUMMARY REPORT       \n");
    printf("========================================\n");
    printf("Total Employees: %d\n", employeeCount);
    if (employeeCount == 0) return;

    double totalSalary = 0.0, highest = employees[0].netSalary, lowest = employees[0].netSalary;
    for (int i = 0; i < employeeCount; i++) {
        totalSalary += employees[i].netSalary;
        if (employees[i].netSalary > highest) highest = employees[i].netSalary;
        if (employees[i].netSalary < lowest) lowest = employees[i].netSalary;
    }
    printf("Average Net Salary : N$%.2f\n", totalSalary / employeeCount);
    printf("Highest Salary     : N$%.2f\n", highest);
    printf("Lowest Salary      : N$%.2f\n", lowest);
}

void generateBudgetReport(void) {
    printf("\n========================================\n");
    printf("           BUDGET SUMMARY REPORT        \n");
    printf("========================================\n");
    double totalAllocated = 0.0, totalSpent = 0.0;
    int overBudgetCount = 0;

    for (int i = 0; i < budgetCount; i++) {
        totalAllocated += budgets[i].allocatedBudget;
        totalSpent += budgets[i].expenditure;
        if (budgets[i].isOverBudget) overBudgetCount++;
    }
    printf("Total Allocated Budget : N$%.2f\n", totalAllocated);
    printf("Total Expenditure      : N$%.2f\n", totalSpent);
    printf("Remaining Balance      : N$%.2f\n", totalAllocated - totalSpent);
    printf("Departments Exceeding Budget: %d\n", overBudgetCount);
}

void generateSupplierReport(void) {
    printf("\n========================================\n");
    printf("          SUPPLIER SUMMARY REPORT       \n");
    printf("========================================\n");
    printf("Total Registered Suppliers: %d\n", supplierCount);
    displaySuppliers();
}

void generateAssetReport(void) {
    printf("\n========================================\n");
    printf("           ASSET SUMMARY REPORT         \n");
    printf("========================================\n");
    printf("Total Registered Assets: %d\n", assetCount);
    double totalVal = 0.0;
    for (int i = 0; i < assetCount; i++) {
        totalVal += assets[i].purchaseValue;
    }
    printf("Total Asset Valuation: N$%.2f\n", totalVal);
    displayAssets();
}

void reportsMenu(void) {
    int choice;
    do {
        printf("\n=== REPORTS MENU ===\n");
        printf("1. Employee Report\n");
        printf("2. Budget Report\n");
        printf("3. Supplier Report\n");
        printf("4. Asset Report\n");
        printf("5. Back to Main Menu\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1: generateEmployeeReport(); break;
            case 2: generateBudgetReport(); break;
            case 3: generateSupplierReport(); break;
            case 4: generateAssetReport(); break;
            case 5: break;
            default: printf("Invalid choice! Try again.\n");
        }
    } while (choice != 5);
}
